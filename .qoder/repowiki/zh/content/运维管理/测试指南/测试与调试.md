### 异常捕获

`python
def safe_dispatch(tool_name: str, args: dict) -> dict:
    """安全的工具分发"""
    try:
        result = registry.dispatch(tool_name, args)
        return {"success": True, "result": result}
    except ToolNotFoundError as e:
        logger.warning(f"Tool not found: {tool_name}")
        return {"success": False, "error": "tool_not_found", "message": str(e)}
    except ValidationError as e:
        logger.warning(f"Validation error: {e}")
        return {"success": False, "error": "validation_error", "message": str(e)}
    except Exception as e:
        logger.error(f"Unexpected error: {e}", exc_info=True)
        return {"success": False, "error": "internal_error", "message": str(e)}
`

import time
from functools import wraps

def timing_decorator(func):
    """计时装饰器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        logger.info(f"{func.__name__} took {elapsed:.3f}s")
        return result
    return wrapper

@timing_decorator

### 性能统计

`python
class PerformanceStats:
    """性能统计"""
    
    def __init__(self):
        self.call_count = 0
        self.total_time = 0.0
        self.min_time = float('inf')
        self.max_time = 0.0
    
    def record(self, duration: float):
        self.call_count += 1
        self.total_time += duration
        self.min_time = min(self.min_time, duration)
        self.max_time = max(self.max_time, duration)
    
    @property
    def avg_time(self):
        return self.total_time / self.call_count if self.call_count > 0 else 0.0
`
