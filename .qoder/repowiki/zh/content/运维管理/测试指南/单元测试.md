    """测试工具注册"""
    # 准备
    tool_name = "test_tool"
    handler = lambda x: x
    schema = {"type": "object", "properties": {}}

    # 验证
    entry = registry.get_entry(tool_name)
    assert entry is not None
    assert entry.name == tool_name
    assert entry.handler == handler

### Mock使用

`python
from unittest.mock import Mock, patch, MagicMock

def test_tool_with_mock_handler():
    """使用模拟处理器测试"""
    mock_handler = Mock(return_value={"result": "mocked"})
    
    registry.register(
        name="mock_tool",
        handler=mock_handler,
        schema={"type": "object", "properties": {}}
    )
    
    result = registry.dispatch("mock_tool", {"input": "test"})
    
    # 验证模拟调用
    mock_handler.assert_called_once_with({"input": "test"})
    assert result["result"] == "mocked"
`

## 测试数据管理

### 测试工厂

`python
class ToolFactory:
    """工具工厂"""
    
    @staticmethod
    def create(name="test_tool", **kwargs):
        return {
            "name": name,
            "description": kwargs.get("description", "测试工具"),
            "parameters": kwargs.get("parameters", {
                "type": "object",
                "properties": {}
            }),
            "handler": kwargs.get("handler", lambda x: x)
        }
    
    @staticmethod
    def create_batch(count=5):
        return [ToolFactory.create(f"tool_{i}") for i in range(count)]
`
