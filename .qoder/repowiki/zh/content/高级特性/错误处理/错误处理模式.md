def should_retry(error: ClassifiedError, attempt: int) -> bool:
    """判断是否应该重试"""
    if not error.retryable:
        return False
    if attempt >= MAX_RETRIES:
        return False
    return True

def get_retry_delay(error: ClassifiedError, attempt: int) -> float:
    """获取重试延迟时间"""
    if error.reason == FailoverReason.RATE_LIMIT:
        return adaptive_rate_limit_backoff(error.provider, attempt)
    return jittered_backoff(attempt)

## 最佳实践

### 错误处理原则

1. **快速失败**: 对于不可重试错误立即返回
2. **智能重试**: 对于可重试错误使用指数退避
3. **降级处理**: 在多次失败后考虑降级方案
4. **监控告警**: 及时发现和处理异常模式

### 错误消息设计

`python
def format_error_message(error: ClassifiedError) -> str:
    """格式化用户友好的错误消息"""
    if error.reason == FailoverReason.RATE_LIMIT:
        return "请求过于频繁，请稍后重试"
    elif error.reason == FailoverReason.CONTEXT_OVERFLOW:
        return "上下文过长，请压缩对话历史"
    elif error.reason == FailoverReason.AUTH:
        return "认证失败，请检查API密钥"
    return f"请求失败: {error.message}"
`
